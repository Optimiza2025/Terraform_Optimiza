server {
    listen 80 default_server;
    listen [::]:80 default_server;

    server_name localhost;

    location /static/ {
        alias /home/ubuntu/Poc-atendimento/optimiza/staticfiles/;
        autoindex on;
    }

        index index.html;
        try_files $uri $uri/ =404;

        include proxy_params;
